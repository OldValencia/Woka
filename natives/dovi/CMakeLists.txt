cmake_minimum_required(VERSION 3.20)
project(woka_dovi C)

set(FFMPEG_SOURCE_DIR "" CACHE PATH "Official FFmpeg 8.1.2 source tree")
if(NOT EXISTS "${FFMPEG_SOURCE_DIR}/libavutil/dovi_meta.h")
    message(FATAL_ERROR "FFMPEG_SOURCE_DIR must contain FFmpeg 8.1.2 headers")
endif()

# Only pixfmt.h's public endian aliases require this generated FFmpeg header.
# No FFmpeg implementation is compiled; detect the target byte order with CMake.
include(TestBigEndian)
test_big_endian(WOKA_BIG_ENDIAN)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/include/libavutil")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/include/libavutil/avconfig.h"
    "#define AV_HAVE_BIGENDIAN ${WOKA_BIG_ENDIAN}\n")

add_library(woka_dovi SHARED woka_dovi.c woka_dovi_metadata.c woka_p5.c)
target_compile_features(woka_dovi PRIVATE c_std_11)
if(NOT WIN32)
    target_link_libraries(woka_dovi PRIVATE m)
endif()
target_include_directories(woka_dovi PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/include")
# FFmpeg's public metadata types use anonymous unions (MSVC C4201). Keep /WX
# for the shim while treating the unmodified upstream headers as external.
target_include_directories(woka_dovi SYSTEM PRIVATE "${FFMPEG_SOURCE_DIR}")
set_target_properties(woka_dovi PROPERTIES C_VISIBILITY_PRESET hidden)
if(MSVC)
    set_property(TARGET woka_dovi PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(woka_dovi PRIVATE /W4 /WX)
else()
    target_compile_options(woka_dovi PRIVATE -Wall -Wextra -Werror)
endif()

if(MSVC AND CMAKE_SYSTEM_PROCESSOR MATCHES "^(AMD64|amd64|x86_64)$")
    add_library(woka_p5_simd STATIC woka_p5_simd.c)
    set_property(TARGET woka_p5_simd PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    target_compile_options(woka_p5_simd PRIVATE /arch:AVX2)
    target_sources(woka_dovi PRIVATE woka_p5_cpu.c)
    target_link_libraries(woka_dovi PRIVATE woka_p5_simd)
    target_compile_definitions(woka_dovi PRIVATE WOKA_P5_SIMD)
    enable_language(CXX)
    file(READ "${CMAKE_CURRENT_SOURCE_DIR}/woka_p5_gpu.hlsl" WOKA_P5_SHADER)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/include/woka_p5_shader.h"
        "static const char WOKA_P5_SHADER[] = R\"WOKA(${WOKA_P5_SHADER})WOKA\";\n")
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS woka_p5_gpu.hlsl)
    target_sources(woka_dovi PRIVATE woka_p5_gpu.cpp)
    target_compile_features(woka_dovi PRIVATE cxx_std_17)
    target_compile_definitions(woka_dovi PRIVATE WOKA_P5_D3D11)
    target_link_libraries(woka_dovi PRIVATE d3d11 d3dcompiler)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(woka_p5_compare tests/p5_compare.c)
    target_include_directories(woka_p5_compare PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
    add_test(NAME dovi_colour_math COMMAND woka_p5_compare --self-test)
    add_executable(woka_p5_benchmark tests/p5_benchmark.c)
    if(TARGET woka_p5_simd)
        set_property(TARGET woka_p5_compare woka_p5_benchmark PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
        target_sources(woka_p5_benchmark PRIVATE woka_p5_cpu.c)
        target_sources(woka_p5_compare PRIVATE woka_p5_cpu.c)
        target_link_libraries(woka_p5_benchmark PRIVATE woka_p5_simd)
        target_link_libraries(woka_p5_compare PRIVATE woka_p5_simd)
        target_compile_definitions(woka_p5_benchmark PRIVATE WOKA_P5_SIMD)
        target_compile_definitions(woka_p5_compare PRIVATE WOKA_P5_SIMD)
    endif()
    find_package(OpenMP QUIET)
    if(OpenMP_C_FOUND)
        target_link_libraries(woka_p5_benchmark PRIVATE OpenMP::OpenMP_C)
        if(TARGET woka_p5_simd)
            add_library(woka_p5_probe SHARED tests/p5_probe.c woka_p5_cpu.c woka_dovi_metadata.c)
            set_property(TARGET woka_p5_probe PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
            target_include_directories(woka_p5_probe PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}"
                "${CMAKE_CURRENT_BINARY_DIR}/include")
            target_include_directories(woka_p5_probe SYSTEM PRIVATE "${FFMPEG_SOURCE_DIR}")
            target_link_libraries(woka_p5_probe PRIVATE woka_p5_simd OpenMP::OpenMP_C)
        endif()
    endif()
    if(NOT WIN32)
        target_link_libraries(woka_p5_compare PRIVATE m)
        target_link_libraries(woka_p5_benchmark PRIVATE m)
    endif()
    add_executable(woka_dovi_metadata_test tests/metadata_test.c)
    target_include_directories(woka_dovi_metadata_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}"
        "${CMAKE_CURRENT_BINARY_DIR}/include")
    target_include_directories(woka_dovi_metadata_test SYSTEM PRIVATE "${FFMPEG_SOURCE_DIR}")
    target_link_libraries(woka_dovi_metadata_test PRIVATE woka_dovi)
    add_test(NAME dovi_metadata_bounds COMMAND woka_dovi_metadata_test)
    add_executable(woka_p5_api_test tests/p5_api_test.c)
    target_include_directories(woka_p5_api_test PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}"
        "${CMAKE_CURRENT_BINARY_DIR}/include")
    target_include_directories(woka_p5_api_test SYSTEM PRIVATE "${FFMPEG_SOURCE_DIR}")
    target_link_libraries(woka_p5_api_test PRIVATE woka_dovi)
    add_test(NAME dovi_colour_api COMMAND woka_p5_api_test)
endif()
